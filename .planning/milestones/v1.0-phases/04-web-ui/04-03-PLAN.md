**5. src/ui/pages/summaries.tsx** — Master-Detail per D-03:
```typescript
export function Component() {
  // Use URL param for selected date
  const { date } = useParams();
  const navigate = useNavigate();
  const { data, isLoading } = useSummaryDates();

  // Auto-select most recent date if none selected
  useEffect(() => {
    if (!date && data?.dates?.length) navigate(`/summaries/${data.dates[0]}`, { replace: true });
  }, [date, data]);

  return (
    <div className="flex h-full">
      {/* Left panel: 280px fixed width, border-right */}
      <div className="w-[280px] min-w-[280px] border-r border-border flex flex-col">
        <div className="p-4 border-b border-border">
          <h1 className="text-xl font-semibold">Summaries</h1>
        </div>
        <DateList dates={data?.dates ?? []} selectedDate={date} onSelect={(d) => navigate(`/summaries/${d}`)} isLoading={isLoading} />
      </div>
      {/* Right panel: fluid */}
      <div className="flex-1 overflow-auto">
        <SummaryDetail date={date} />
      </div>
    </div>
  );
}
```

Responsive: at `<768px`, show only the DateList (full width) when no date selected, or only SummaryDetail (full width with back button) when date selected.
